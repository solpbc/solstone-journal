cmake_minimum_required(VERSION 3.18)
project(parakeet_error_retirement_control LANGUAGES C CXX)
if(NOT EXISTS "${PARAKEET_SOURCE}/src/ggml_graph.cpp" OR NOT EXISTS "${GGML_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Explicit pinned patched Parakeet and GGML sources are required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml)
find_package(Threads REQUIRED)
function(add_retirement_control target root control)
    add_executable(${target} "${control}" "${root}/src/model_loader.cpp"
        "${root}/src/backend_pool.cpp" "${root}/src/silero_vad.cpp" "${root}/src/bundle.cpp")
    target_include_directories(${target} PRIVATE "${root}/src" "${root}/include")
    target_compile_definitions(${target} PRIVATE NDEBUG
        PARAKEET_TEST_BACKEND_SOURCE="${root}/src/backend.cpp"
        PARAKEET_TEST_GRAPH_SOURCE="${root}/src/ggml_graph.cpp")
    target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections)
    target_link_libraries(${target} PRIVATE ggml Threads::Threads)
    if(APPLE)
        target_link_options(${target} PRIVATE -Wl,-dead_strip)
    else()
        target_link_options(${target} PRIVATE -Wl,--gc-sections)
    endif()
endfunction()
add_retirement_control(parakeet_error_retirement_control "${PARAKEET_SOURCE}" control.cpp)
if(EXISTS "${PARAKEET_BASELINE_SOURCE}/src/ggml_graph.cpp")
    add_retirement_control(parakeet_error_retirement_baseline "${PARAKEET_BASELINE_SOURCE}" old_control.cpp)
endif()
if(EXISTS "${PARAKEET_ANDROID_SOURCE}/src/ggml_graph.cpp")
    add_retirement_control(parakeet_error_retirement_android_shape "${PARAKEET_ANDROID_SOURCE}" control.cpp)
    target_compile_definitions(parakeet_error_retirement_android_shape PRIVATE PARAKEET_TEST_HAS_THREAD_RELEASE)
endif()
enable_testing()
add_test(NAME parakeet_error_retirement_control COMMAND parakeet_error_retirement_control)
set_tests_properties(parakeet_error_retirement_control PROPERTIES TIMEOUT 15 ENVIRONMENT "PARAKEET_DEVICE=cpu")
if(TARGET parakeet_error_retirement_android_shape)
    add_test(NAME parakeet_error_retirement_android_shape COMMAND parakeet_error_retirement_android_shape)
    set_tests_properties(parakeet_error_retirement_android_shape PROPERTIES TIMEOUT 15 ENVIRONMENT "PARAKEET_DEVICE=cpu")
endif()
