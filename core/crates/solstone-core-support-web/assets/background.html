let badgeTask = null; AppServices.register('support', { initialize() { badgeTask = AppServices.registerTask('support', 'badge-count', { intervalMs: 5 * 60 * 1000, run: async ({ apiJson }) => apiJson('/app/support/api/badge-count'), onSuccess: (data) => { AppServices.badges.app.set('support', data.count ?? 0); } }); }, stop() { if (badgeTask) { badgeTask.stop(); badgeTask = null; } } });
